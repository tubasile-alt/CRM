## Objetivo

A aba `Botox` passa a ser um espelho fiel do banco, reconstruído por
reconciliação. Hoje ela é alimentada por `append` disparado em dois pontos
legados do `app.py`, dentro de uma thread daemon — o que faz linhas
desaparecerem silenciosamente e explica a aba parar em março/2026.

**Não depende de `MESSAGE_DISPATCH_ENABLED`.** A planilha funciona com a flag
desligada; as colunas de status simplesmente aparecem vazias até a Fase 5.

## Aviso: Autoscale escala a zero

O deploy é `deploymentTarget = "autoscale"`. O container é encerrado quando não
há tráfego. Um `BackgroundScheduler` só dispara enquanto o processo vive —
portanto **o job de 6h é best-effort, não garantia**.

Isso já vale para o scheduler de auto-faltou existente (`app.py:4152`).

Por isso a lógica fica em função pura, invocável de três formas:

| Gatilho | Confiabilidade | Fase |
|---|---|---|
| `python scripts/sync_botox_sheet.py` | manual, 100% | esta |
| APScheduler 6h | best-effort | esta |
| n8n → endpoint com API key | garantida | Fase 4 |

Não tente resolver a confiabilidade nesta fase.

### 1b. Escrita síncrona da aba

Adicionar. **Síncrona de propósito** — sem `threading.Thread`. Quem chama decide
o contexto de execução, e o erro precisa subir.

```python
def write_botox_sheet(values_matrix):
    """Reescreve a aba Botox inteira. values_matrix já inclui o cabeçalho.

    Ordem deliberada: escreve primeiro, limpa a sobra depois. Se a escrita
    falhar, a planilha permanece com o conteúdo anterior — nunca vazia.

    Retorna (ok: bool, mensagem: str). Não levanta exceção.
    """
    try:
        import requests as req
        token = _get_access_token()
        h = {'Authorization': f'Bearer {token}',
             'Content-Type': 'application/json'}
        base = f'https://sheets.googleapis.com/v4/spreadsheets/{BOTOX_SPREADSHEET_ID}'

        class _S:
            def get(self, *a, **kw):  return req.get(*a, headers=h, **kw)
            def post(self, *a, **kw): return req.post(*a, headers=h, **kw)
            def put(self, *a, **kw):  return req.put(*a, headers=h, **kw)

        if not _ensure_botox_sheet(_S(), base, token):
            return False, 'não foi possível garantir a aba Botox'

        resp = req.put(
            f'{base}/values/{BOTOX_SHEET_NAME}!A1',
            headers=h,
            timeout=60,
            params={'valueInputOption': 'USER_ENTERED'},
            json={'values': values_matrix},
        )
        if resp.status_code not in (200, 201):
            return False, f'PUT falhou: {resp.status_code} {resp.text[:200]}'

        # Limpa apenas o que sobrou abaixo do bloco recém-escrito.
        first_stale_row = len(values_matrix) + 1
        req.post(
            f'{base}/values/{BOTOX_SHEET_NAME}!A{first_stale_row}:I100000:clear',
            headers=h, timeout=30,
        )

        return True, f'{len(values_matrix) - 1} linhas sincronizadas'

    except Exception as e:
        return False, f'erro inesperado: {e}'
```

## Passo 2 — `services/botox_sheet_service.py` (arquivo novo)

```python
"""Reconciliação da aba Botox do Google Sheets a partir do banco.

A planilha é um espelho de leitura: reconstruída por completo a cada execução.
Nenhum caminho de request escreve nela.
"""
import os
from sqlalchemy import text
from sqlalchemy.orm import aliased

ADVISORY_LOCK_KEY = 918273


def build_botox_sheet_rows():
    """Retorna a matriz de valores da aba, cabeçalho incluído."""
    from models import (db, ProcedureExecution, CosmeticProcedurePlan,
                        Note, Patient, MessageDispatch)
    from services.google_sheets import BOTOX_HEADERS, format_phone_for_sheets

    d0 = aliased(MessageDispatch)
    m5 = aliased(MessageDispatch)

    rows = (
        db.session.query(ProcedureExecution, Patient, d0, m5)
        .join(CosmeticProcedurePlan,
              ProcedureExecution.plan_id == CosmeticProcedurePlan.id)
        .join(Note, CosmeticProcedurePlan.note_id == Note.id)
        .join(Patient, Note.patient_id == Patient.id)
        .outerjoin(d0, db.and_(d0.execution_id == ProcedureExecution.id,
                               d0.message_type == 'd0'))
        .outerjoin(m5, db.and_(m5.execution_id == ProcedureExecution.id,
                               m5.message_type == 'm5'))
        .filter(
            ProcedureExecution.execution_status == 'realizada',
            ProcedureExecution.performed_date.isnot(None),
            db.func.lower(CosmeticProcedurePlan.procedure_name).like('%botox%'),
        )
        .order_by(ProcedureExecution.performed_date.asc())
        .all()
    )

    def _d(value):
        return value.strftime('%d/%m/%Y') if value else ''

    def _dt(value):
        return value.strftime('%d/%m/%Y %H:%M') if value else ''

    matrix = [list(BOTOX_HEADERS)]
    for execution, patient, disp_d0, disp_m5 in rows:
        matrix.append([
            execution.id,
            patient.name or '',
            format_phone_for_sheets(patient.phone),
            _d(execution.performed_date),
            _d(execution.followup_date),
            disp_d0.status if disp_d0 else '',
            _dt(disp_d0.sent_at) if disp_d0 else '',
            disp_m5.status if disp_m5 else '',
            _dt(disp_m5.sent_at) if disp_m5 else '',
        ])
    return matrix


def run_botox_sheet_sync():
    """Ponto de entrada único. Seguro para chamadas concorrentes.

    Retorna (ok: bool, mensagem: str).
    """
    from models import db
    from services.google_sheets import write_botox_sheet

    is_postgres = db.engine.dialect.name == 'postgresql'

    if not is_postgres:
        matrix = build_botox_sheet_rows()
        return write_botox_sheet(matrix)

    # Advisory lock: o gunicorn roda --workers=2 e ambos sobem o scheduler.
    # A conexão precisa ser mantida aberta durante todo o lock.
    with db.engine.connect() as conn:
        acquired = conn.execute(
            text('SELECT pg_try_advisory_lock(:k)'), {'k': ADVISORY_LOCK_KEY}
        ).scalar()

        if not acquired:
            return True, 'ignorado: outro worker já está sincronizando'

        try:
            matrix = build_botox_sheet_rows()
            return write_botox_sheet(matrix)
        finally:
            conn.execute(text('SELECT pg_advisory_unlock(:k)'),
                         {'k': ADVISORY_LOCK_KEY})
```

Detalhe que não pode ser simplificado: o advisory lock do PostgreSQL é
**por sessão**. Com o pool do SQLAlchemy, `db.session.execute()` pode pegar
conexões diferentes entre o lock e o unlock, e o lock vaza. Daí o
`db.engine.connect()` explícito segurando a mesma conexão do início ao fim.

## Passo 3 — `scripts/sync_botox_sheet.py` (arquivo novo)

```python
"""Sincroniza a aba Botox sob demanda.

Uso:
    python scripts/sync_botox_sheet.py            # sincroniza
    python scripts/sync_botox_sheet.py --dry-run  # só mostra o que faria
"""
import sys

from app import app
from services.botox_sheet_service import build_botox_sheet_rows, run_botox_sheet_sync


def main():
    dry_run = '--dry-run' in sys.argv

    with app.app_context():
        if dry_run:
            matrix = build_botox_sheet_rows()
            print(f'linhas de dados: {len(matrix) - 1}')
            print(f'cabeçalho: {matrix[0]}')
            for row in matrix[1:4]:
                print(f'  {row}')
            if len(matrix) > 4:
                print(f'  ... (+{len(matrix) - 4})')
            print('\nDRY-RUN: nada foi escrito na planilha.')
            return

        ok, msg = run_botox_sheet_sync()
        print(('✓ ' if ok else '✗ ') + msg)
        sys.exit(0 if ok else 1)


if __name__ == '__main__':
    main()
```
