"""Integração SOMENTE LEITURA com o CapillaryTracker (ICB) via HTTP.

Busca, pelo nome do paciente, as cirurgias registradas no Tracker.
O CRM nunca acessa o banco do Tracker: chama a rota
/api/integracao/cirurgias do Tracker, protegida por chave.

Configuração (Secrets do Replit do CRM):
  TRACKER_API_URL  ex.: https://seu-tracker.replit.app
  TRACKER_API_KEY  mesma chave definida em INTEGRATION_API_KEY no Tracker
"""
import json
import logging
import os
import re
import unicodedata
import urllib.error
import urllib.parse
import urllib.request
from difflib import SequenceMatcher

logger = logging.getLogger(__name__)

TIMEOUT_SECONDS = 12
MAX_RESULTS = 30


class TrackerNotConfigured(RuntimeError):
    """URL/chave do Tracker ausentes ou recusadas pelo Tracker."""


def _config():
    base = (os.environ.get('TRACKER_API_URL') or '').strip().rstrip('/')
    key = (os.environ.get('TRACKER_API_KEY') or '').strip()
    if not base or not key:
        raise TrackerNotConfigured('TRACKER_API_URL / TRACKER_API_KEY não configuradas')
    host = urllib.parse.urlparse(base).hostname or ''
    if not base.startswith('https://') and host not in ('localhost', '127.0.0.1'):
        raise TrackerNotConfigured('TRACKER_API_URL deve usar https://')
    return base, key


def normalize_name(name):
    """minúsculas, sem acento, sem pontuação, espaços colapsados."""
    if not name:
        return ''
    s = unicodedata.normalize('NFKD', str(name))
    s = ''.join(c for c in s if not unicodedata.combining(c))
    s = re.sub(r'[^a-zA-Z0-9\s]', ' ', s).lower()
    return re.sub(r'\s+', ' ', s).strip()


def score_names(a_norm, b_norm):
    """Similaridade 0..1 entre dois nomes já normalizados."""
    if not a_norm or not b_norm:
        return 0.0
    if a_norm == b_norm:
        return 1.0
    ta, tb = set(a_norm.split()), set(b_norm.split())
    jaccard = len(ta & tb) / len(ta | tb)
    seq = SequenceMatcher(
        None, ' '.join(sorted(ta)), ' '.join(sorted(tb))
    ).ratio()
    score = max(jaccard, seq)
    # Um nome contido no outro (ex.: "Maria Silva" x "Maria Souza Silva")
    if min(len(ta), len(tb)) >= 2 and (ta <= tb or tb <= ta):
        score = max(score, 0.9)
    return round(score, 3)


def _label(score):
    if score >= 1.0:
        return 'exato'
    if score >= 0.85:
        return 'provavel'
    return 'possivel'


def _fetch_candidates(nome):
    base, key = _config()
    url = f"{base}/api/integracao/cirurgias?" + urllib.parse.urlencode({'nome': nome})
    req = urllib.request.Request(
        url, headers={'X-API-Key': key, 'Accept': 'application/json'}
    )
    try:
        with urllib.request.urlopen(req, timeout=TIMEOUT_SECONDS) as resp:
            payload = json.loads(resp.read().decode('utf-8'))
    except urllib.error.HTTPError as e:
        if e.code in (401, 503):
            # chave errada no CRM ou INTEGRATION_API_KEY ausente no Tracker
            raise TrackerNotConfigured(f'Tracker recusou a chave (HTTP {e.code})')
        raise
    return payload.get('matches') or []


def search_by_name(nome):
    """Cirurgias do Tracker cujo nome casa com `nome`.

    Retorna lista de dicts (todas as colunas de `surgery`) acrescida de:
      match_score (0..1), match_label (exato|provavel|possivel), necroses (lista).
    Ordenada por similaridade e, depois, pela data mais recente.
    """
    target = normalize_name(nome)
    if not [t for t in target.split() if len(t) >= 2]:
        return []

    results = []
    for d in _fetch_candidates(nome):
        score = score_names(target, normalize_name(d.get('nome')))
        if score < 0.5:
            continue
        d['match_score'] = score
        d['match_label'] = _label(score)
        d.setdefault('necroses', [])
        results.append(d)

    # maior similaridade primeiro; no empate, data mais recente primeiro
    results.sort(key=lambda x: (x['match_score'], x.get('data') or ''), reverse=True)
    return results[:MAX_RESULTS]
